# seller.secrets.txt — the secrets this repo's services need.
#
# `usvc_seller secrets upload seller.secrets.txt` seeds both the value (used for
# gateway-side testing) and the comment-block description shown to customers
# (unitysvc/unitysvc#1618). Values are env-only — `${NAME}` with no default —
# so the real key must come from the environment: a GitHub Actions secret in CI,
# or your shell locally.
#
# Sourceable locally:  set -a; . ./seller.secrets.txt; set +a

# AWS_BEARER_TOKEN_BEDROCK — Amazon Bedrock API key.
#   Your Amazon Bedrock API key. Generate one in the AWS console under Amazon
#   Bedrock → API keys (it is used as a bearer token; AWS documents it as the
#   AWS_BEARER_TOKEN_BEDROCK environment variable). The key is scoped to your AWS
#   account and its enabled Bedrock model access; make sure the models you call
#   are access-granted in your account's region.
#
#   Customers bring their own key under this same name as a customer secret for
#   the OpenAI-compatible mantle channel (byok); the seller secret of the same
#   name is what the platform's gateway-side tests authenticate with.
AWS_BEARER_TOKEN_BEDROCK=${AWS_BEARER_TOKEN_BEDROCK}

# AWS_ACCESS_KEY_ID — AWS IAM access key id with Amazon Bedrock permissions.
#   An IAM access key for an identity allowed to call Amazon Bedrock runtime
#   operations (bedrock:InvokeModel / Converse) in your account. Used by the
#   native Bedrock runtime channel (converse): the gateway signs upstream
#   requests with SigV4 from these keys, so boto3 Converse/InvokeModel works
#   through the gateway; pairs with AWS_SECRET_ACCESS_KEY.
AWS_ACCESS_KEY_ID=${AWS_ACCESS_KEY_ID}

# AWS_SECRET_ACCESS_KEY — AWS IAM secret access key paired with
#   AWS_ACCESS_KEY_ID above. Never shared with the upstream provider beyond
#   request signing; stored encrypted on the platform.
AWS_SECRET_ACCESS_KEY=${AWS_SECRET_ACCESS_KEY}
